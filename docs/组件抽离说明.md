### 结构

> ESM

```
- packages
  |- core/
  |- titlebar/
  |   |- src/
  |   |   |- core/
  |   |   |- main/
  |   |   |   └─ index.ts
  |   |   |- render/
  |   |   |   └─ index.ts
  |   |   └─ preload/
  |   |       └─ index.ts
  |   └─ README.md

```

**import**

```js
import { expose } from '@hocgin/titlebar/preload' /// preload
import { use } from '@hocgin/titlebar/main' /// main
import { titlebar } from '@hocgin/titlebar/renderer' /// renderer

```

**element.jsx**

```ts
/// <reference types="@hocgin/titlebar/renderer" />
import { HTMLTitleBarElementAttributes } from '@hocgin/titlebar/renderer'

declare global {
  namespace JSX {
    interface IntrinsicElements {
      // Define an interface for the web component props
      'title-bar': Partial<HTMLTitleBarElementAttributes>
    }
  }
}

// or
import { ElectronAPI } from '@hocgin/toolkit/preload'

declare global {
  interface Window {
    electron: ElectronAPI
  }
}

```
